{{/* 构建时盘古化（替代运行时 pangu.js）。 做法：先用占位符把所有 HTML 标签整体摘出，对纯文本做中英文加空格， 再把标签还原。这样跨标签的中英相邻（如 **Tumbleweed**和**Leap** ） 也能正确处理，且标签属性（id/class/href）完全不受影响。 */}} {{- $content := .Content -}} {{- $tags := findRE `<[^>]*>` $content -}} {{- $masked := $content -}} {{- range $i, $t := $tags -}} {{- $masked = replace $masked $t (printf "@@TAG%d@@" $i) -}} {{- end -}} {{- $masked = replaceRE `([一-鿿，。！？：；、（）【】「」《》“”‘’])([a-zA-Z0-9])` `${1} ${2}` $masked -}} {{- $masked = replaceRE `([a-zA-Z0-9])([一-鿿，。！？：；、（）【】「」《》“”‘’])` `${1} ${2}` $masked -}} {{- range $i, $t := $tags -}} {{- $masked = replace $masked (printf "@@TAG%d@@" $i) $t -}} {{- end -}} {{- $masked | safeHTML -}}
